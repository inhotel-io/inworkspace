# inWorkspace Console Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the read-only inWorkspace Console described in [`design.md`](design.md): agent conversations read live over JMAP, server events received through the signed event webhook, and a 24-hour overview.

**Architecture:** One Next.js 16 app under `console/`, run as two roles from one image. The `ingest` role verifies and stores webhook batches in Postgres; the `web` role sits behind an identity-aware proxy and renders pages from Postgres and from JMAP reads made as the agent (master-user login). Pure units (`config`, `json`, `signature`, `normalize`, `conversations`, `sanitize`, `auth`, …) carry the logic and the tests; pages and routes are thin.

**Tech Stack:** Node 24, Next.js 16.4 (App Router, Cache Components, `proxy.ts`), React 19, TypeScript 5, Tailwind CSS 4, `pg`, `jose`, `sanitize-html`, Vitest 5, Docker, Stalwart v0.16 for end-to-end runs.

**Facts this plan relies on** (verified in a spike against `stalwartlabs/stalwart:v0.16.16`):

- Registry methods use the capability `urn:stalwart:jmap`; lists in registry objects are maps keyed by integer strings (`{"0": {...}}`); durations are milliseconds; back-references (`#ids`) work; `x:Account/query` ignores `position`, so the directory reads each domain in one query.
- Master-user login is `<mailbox>%<master>` with the master's password; the session's `primaryAccounts["urn:ietf:params:jmap:mail"]` is the mailbox's account.
- `EmailSubmission/set` validates recipient domains against the public suffix list, so tests use `example.com` (IANA-reserved, null MX: nothing is ever delivered).
- Delivery status notifications carry `Auto-Submitted: auto-generated` and come from `MAILER-DAEMON@…`.
- `x:Action/set` with `{"@type": "ReloadSettings"}` fails on v0.16.16 with a validation error in a default DNSBL setting; webhooks then activate on restart.
- Webhook bodies contain 64-bit integers (`queueId`, `spanId`); `JSON.parse` with source-text access (Node 24) keeps them exact.

```ts
export type ConsoleRole = "web" | "ingest" | "all";
export type AuthMode = "iap" | "none";

export interface ConsoleConfig {
  role: ConsoleRole;
  databaseUrl: string;
  /** Ingest role only; empty otherwise. */
  webhookSigningKey: string;
  retentionDays: number;
  /** Web role only; empty otherwise. */
  jmapUrl: string;
  jmapPrincipal: string;
  jmapPrincipalSecret: string;
  agentDomains: string[];
  agentLocalpartPattern: RegExp | null;
  authMode: AuthMode;
  iapAudience: string;
  iapJwksUrl: string;
  allowedEmails: string[];
  allowedDomains: string[];
  displayTimeZone: string;
}

export class ConfigError extends Error {
  constructor(readonly problems: string[]) {
    super(`Invalid console configuration: ${problems.join("; ")}`);
    this.name = "ConfigError";
  }
}

type Env = Readonly<Record<string, string | undefined>>;

const GOOGLE_IAP_JWKS_URL = "https://www.gstatic.com/iap/verify/public_key-jwk";

export function servesIngest(config: Pick<ConsoleConfig, "role">): boolean {
  return config.role === "ingest" || config.role === "all";
}

export function servesWeb(config: Pick<ConsoleConfig, "role">): boolean {
  return config.role === "web" || config.role === "all";
}

function text(env: Env, name: string): string {
  return env[name]?.trim() ?? "";
}

function list(env: Env, name: string): string[] {
  return text(env, name)
    .split(",")
    .map((item) => item.trim().toLowerCase())
    .filter((item) => item.length > 0);
}

export function parseConfig(env: Env): ConsoleConfig {
  const problems: string[] = [];
  const need = (name: string): string => {
    const value = text(env, name);
    if (!value) problems.push(`${name} is required`);
    return value;
  };

  const role = text(env, "CONSOLE_ROLE") as ConsoleRole;
  if (role !== "web" && role !== "ingest" && role !== "all") {
    problems.push("CONSOLE_ROLE must be web, ingest or all");
  }
  const databaseUrl = need("DATABASE_URL");

  let webhookSigningKey = "";
  let retentionDays = 30;
  if (servesIngest({ role })) {
    webhookSigningKey = need("WEBHOOK_SIGNING_KEY");
    if (webhookSigningKey && webhookSigningKey.length < 32) {
      problems.push("WEBHOOK_SIGNING_KEY must be at least 32 characters");
    }
    const days = text(env, "EVENT_RETENTION_DAYS");
    if (days) {
      retentionDays = Number(days);
      if (!Number.isInteger(retentionDays) || retentionDays < 1) {
        problems.push("EVENT_RETENTION_DAYS must be a positive integer");
      }
    }
  }

  let jmapUrl = "";
  let jmapPrincipal = "";
  let jmapPrincipalSecret = "";
  let agentDomains: string[] = [];
  let agentLocalpartPattern: RegExp | null = null;
  let authMode: AuthMode = "iap";
  let iapAudience = "";
  let iapJwksUrl = GOOGLE_IAP_JWKS_URL;
  let allowedEmails: string[] = [];
  let allowedDomains: string[] = [];
  let displayTimeZone = "UTC";
  if (servesWeb({ role })) {
    jmapUrl = need("JMAP_URL").replace(/\/+$/, "");
    if (jmapUrl && !/^https?:\/\/[^/]+$/.test(jmapUrl)) {
      problems.push("JMAP_URL must be an http(s) origin such as https://mail.example.com");
    }
    jmapPrincipal = need("JMAP_PRINCIPAL").toLowerCase();
    jmapPrincipalSecret = need("JMAP_PRINCIPAL_SECRET");
    agentDomains = list(env, "AGENT_DOMAINS");
    if (agentDomains.length === 0) problems.push("AGENT_DOMAINS is required");

    const pattern = text(env, "AGENT_LOCALPART_PATTERN");
    if (pattern) {
      try {
        agentLocalpartPattern = new RegExp(`^(?:${pattern})$`);
      } catch {
        problems.push("AGENT_LOCALPART_PATTERN is not a valid regular expression");
      }
    }

    const mode = text(env, "AUTH_MODE") || "iap";
    if (mode !== "iap" && mode !== "none") problems.push("AUTH_MODE must be iap or none");
    authMode = mode === "none" ? "none" : "iap";
    if (mode === "iap") {
      iapAudience = need("IAP_AUDIENCE");
      iapJwksUrl = text(env, "IAP_JWKS_URL") || GOOGLE_IAP_JWKS_URL;
      allowedEmails = list(env, "ALLOWED_EMAILS");
      allowedDomains = list(env, "ALLOWED_DOMAINS");
      if (allowedEmails.length === 0 && allowedDomains.length === 0) {
        problems.push("ALLOWED_EMAILS or ALLOWED_DOMAINS is required when AUTH_MODE=iap");
      }
    } else if (mode === "none" && env.NODE_ENV === "production" && env.ALLOW_INSECURE_AUTH_NONE !== "1") {
      problems.push("AUTH_MODE=none is refused in production unless ALLOW_INSECURE_AUTH_NONE=1");
    }

    displayTimeZone = text(env, "DISPLAY_TIME_ZONE") || "UTC";
    try {
      new Intl.DateTimeFormat("en-GB", { timeZone: displayTimeZone });
    } catch {
      problems.push("DISPLAY_TIME_ZONE is not a valid IANA time zone");
    }
  }

  if (problems.length > 0) throw new ConfigError(problems);
  return {
    role,
    databaseUrl,
    webhookSigningKey,
    retentionDays,
    jmapUrl,
    jmapPrincipal,
    jmapPrincipalSecret,
    agentDomains,
    agentLocalpartPattern,
    authMode,
    iapAudience,
    iapJwksUrl,
    allowedEmails,
    allowedDomains,
    displayTimeZone,
  };
}

let cached: ConsoleConfig | undefined;

/** The configuration of this process, parsed on first use. Never call at module top level: builds run without it. */
export function getConfig(): ConsoleConfig {
  cached ??= parseConfig(process.env);
  return cached;
}
```

```ts
/**
 * JSON.parse, except that integers too large for a double — the server's 64-bit queue and
 * span ids — come back as their exact decimal strings instead of rounded numbers.
 */
export function parseJsonExact(text: string): unknown {
  return JSON.parse(text, (_key: string, value: unknown, context?: { source?: string }) => {
    if (
      typeof value === "number" &&
      Number.isInteger(value) &&
      !Number.isSafeInteger(value) &&
      context?.source !== undefined
    ) {
      return context.source;
    }
    return value;
  });
}
```

`src/lib/signature.ts`:

```ts
import { createHmac, timingSafeEqual } from "node:crypto";

/** The server's X-Signature: base64 HMAC-SHA256 of the body, keyed with the key's UTF-8 bytes. */
export function signBody(body: Uint8Array | string, key: string): string {
  return createHmac("sha256", key).update(body).digest("base64");
}

export function verifySignature(body: Uint8Array, header: string | null, key: string): boolean {
  if (!header || !key) return false;
  const given = Buffer.from(header.trim(), "base64");
  const expected = createHmac("sha256", key).update(body).digest();
  return given.length === expected.length && timingSafeEqual(given, expected);
}
```

```ts
import { createHash } from "node:crypto";
import { severityOf, type Severity } from "./eventTypes";
import { parseJsonExact } from "./json";

export interface WebhookEvent {
  createdAt: string;
  type: string;
  data: Record<string, unknown>;
}

export interface EventRow {
  dedupeHash: Buffer;
  eventAt: Date;
  type: string;
  severity: Severity;
  sender: string | null;
  recipients: string[];
  addresses: string[];
  remoteIp: string | null;
  queueId: string | null;
  messageId: string | null;
  data: Record<string, unknown>;
}

export class BatchError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "BatchError";
  }
}

const EMAIL_ADDRESS = /^[^\s@<>"]+@[^\s@<>"]+$/;

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

function emailOf(value: unknown): string | null {
  if (typeof value !== "string") return null;
  const address = value.trim().toLowerCase();
  return EMAIL_ADDRESS.test(address) ? address : null;
}

function emailsOf(value: unknown): string[] {
  const values = Array.isArray(value) ? value : [value];
  return [...new Set(values.map(emailOf).filter((address): address is string => address !== null))];
}

function scalarOf(value: unknown): string | null {
  if (typeof value === "string") return value.length > 0 ? value : null;
  if (typeof value === "number") return String(value);
  return null;
}

/** JSON with object keys sorted at every level, so equal data always hashes the same. */
export function canonicalJson(value: unknown): string {
  if (Array.isArray(value)) return `[${value.map(canonicalJson).join(",")}]`;
  if (isRecord(value)) {
    const entries = Object.keys(value)
      .sort()
      .map((key) => `${JSON.stringify(key)}:${canonicalJson(value[key])}`);
    return `{${entries.join(",")}}`;
  }
  return JSON.stringify(value) ?? "null";
}

export function parseBatch(text: string): WebhookEvent[] {
  let parsed: unknown;
  try {
    parsed = parseJsonExact(text);
  } catch {
    throw new BatchError("the body is not valid JSON");
  }
  if (!isRecord(parsed) || !Array.isArray(parsed.events)) {
    throw new BatchError("the body is not an event batch");
  }
  return parsed.events.map((event: unknown, index: number) => {
    if (
      !isRecord(event) ||
      typeof event.type !== "string" ||
      event.type.length === 0 ||
      typeof event.createdAt !== "string" ||
      Number.isNaN(Date.parse(event.createdAt))
    ) {
      throw new BatchError(`event ${index} is malformed`);
    }
    return { createdAt: event.createdAt, type: event.type, data: isRecord(event.data) ? event.data : {} };
  });
}

export function normalizeEvent(event: WebhookEvent): EventRow {
  const { data } = event;
  const sender = emailOf(data.from);
  const recipients = emailsOf(data.to);
  const account = emailOf(data.accountName);
  const addresses = [...new Set([sender, ...recipients, account].filter((address): address is string => address !== null))];
  return {
    dedupeHash: createHash("sha256").update(`${event.createdAt}\n${event.type}\n${canonicalJson(data)}`).digest(),
    eventAt: new Date(event.createdAt),
    type: event.type,
    severity: severityOf(event.type),
    sender,
    recipients,
    addresses,
    remoteIp: scalarOf(data.remoteIp),
    queueId: scalarOf(data.queueId),
    messageId: scalarOf(data.messageId),
    data,
  };
}
```

```ts
  {
    version: "001_events",
    sql: `
      create table events (
        id          bigserial primary key,
        dedupe_hash bytea       not null unique,
        event_at    timestamptz not null,
        received_at timestamptz not null default now(),
        type        text        not null,

        recipients  text[]      not null default '{}',

        queue_id    text,

      );
      create index events_event_at_idx  on events (event_at desc);
      create index events_type_idx      on events (type, event_at desc);
      create index events_sender_idx    on events (sender, event_at desc);
      create index events_addresses_idx on events using gin (addresses);
    `,
  },
];

const MIGRATION_LOCK = 7_302_025;

/** Applies pending migrations in order, under an advisory lock so that two instances never race. */
export async function migrate(pool: Pool): Promise<string[]> {
  const client = await pool.connect();
  try {
    await client.query("select pg_advisory_lock($1)", [MIGRATION_LOCK]);
    await client.query(
      "create table if not exists schema_migrations (version text primary key, applied_at timestamptz not null default now())",
    );
    const { rows } = await client.query<{ version: string }>("select version from schema_migrations");
    const applied = new Set(rows.map((row) => row.version));
    const newlyApplied: string[] = [];
    for (const migration of MIGRATIONS) {
      if (applied.has(migration.version)) continue;
      await client.query("begin");
      try {
        await client.query(migration.sql);
        await client.query("insert into schema_migrations (version) values ($1)", [migration.version]);
        await client.query("commit");
      } catch (error) {
        await client.query("rollback");
        throw error;
      }
      newlyApplied.push(migration.version);
    }
    return newlyApplied;
  } finally {
    await client.query("select pg_advisory_unlock($1)", [MIGRATION_LOCK]).catch(() => undefined);
    client.release();
  }
}
```

```ts
const INSERT_CHUNK = 500;
const COLUMNS = 11;
const SELECT_COLUMNS = "id, event_at, type, severity, sender, recipients, remote_ip, queue_id, message_id, data";

export async function insertEvents(pool: Pool, rows: readonly EventRow[]): Promise<number> {
  const unique = [...new Map(rows.map((row) => [row.dedupeHash.toString("hex"), row])).values()];
  if (unique.length === 0) return 0;
  const client = await pool.connect();
  try {
    await client.query("begin");
    let inserted = 0;
    for (let start = 0; start < unique.length; start += INSERT_CHUNK) {
      inserted += await insertChunk(client, unique.slice(start, start + INSERT_CHUNK));
    }
    await client.query("commit");
    return inserted;
  } catch (error) {
    await client.query("rollback").catch(() => undefined);
    throw error;
  } finally {
    client.release();
  }
}

async function insertChunk(client: PoolClient, rows: readonly EventRow[]): Promise<number> {
  const values: unknown[] = [];
  const tuples = rows.map((row, index) => {
    values.push(
      row.dedupeHash,
      row.eventAt,
      row.type,
      row.severity,
      row.sender,
      row.recipients,
      row.addresses,
      row.remoteIp,
      row.queueId,
      row.messageId,
      JSON.stringify(row.data).replaceAll("\\u0000", ""),
    );
    const p = (column: number) => `$${index * COLUMNS + column}`;
    return `(${p(1)}, ${p(2)}, ${p(3)}, ${p(4)}, ${p(5)}, ${p(6)}::text[], ${p(7)}::text[], ${p(8)}, ${p(9)}, ${p(10)}, ${p(11)}::jsonb)`;
  });
  const result = await client.query(
    `insert into events (dedupe_hash, event_at, type, severity, sender, recipients, addresses, remote_ip, queue_id, message_id, data)
     values ${tuples.join(", ")}
     on conflict (dedupe_hash) do nothing`,
    values,
  );
  return result.rowCount ?? 0;
}

function toStoredEvent(row: Record<string, unknown>): StoredEvent {
  return {

    remoteIp: (row.remote_ip as string | null) ?? null,

function escapeLike(value: string): string {
  return value.replace(/[\\%_]/g, (character) => `\\${character}`);
}

/** Newest first; with `afterId`, the events after it, still returned newest first. */
export async function queryEvents(pool: Pool, query: EventQuery): Promise<StoredEvent[]> {
  const conditions: string[] = [];
  const values: unknown[] = [];
  const where = (template: string, value: unknown) => {
    values.push(value);
    conditions.push(template.replace("?", `$${values.length}`));
  };
  if (query.typePrefix) where("type like ?", `${escapeLike(query.typePrefix)}%`);
  if (query.severity) where("severity = ?", query.severity);
  if (query.address) where("addresses @> array[?]::text[]", query.address.toLowerCase());
  if (query.since) where("event_at >= ?", query.since);
  if (query.beforeId !== undefined) where("id < ?", query.beforeId);
  if (query.afterId !== undefined) where("id > ?", query.afterId);
  const ascending = query.afterId !== undefined;
  values.push(Math.min(Math.max(Math.trunc(query.limit), 1), 500));
  const result = await pool.query(
    `select ${SELECT_COLUMNS} from events
     ${conditions.length > 0 ? `where ${conditions.join(" and ")}` : ""}
     order by id ${ascending ? "asc" : "desc"}
     limit $${values.length}`,
    values,
  );
  const events = result.rows.map(toStoredEvent);
  return ascending ? events.reverse() : events;
}

  pool: Pool,
  params: { since: Date; agentAddresses: readonly string[] },
): Promise<AgentActivity[]> {
  if (params.agentAddresses.length === 0) return [];
  const result = await pool.query(
    `select a.address,
       count(e.id) filter (where e.type = 'message-ingest.ham' and a.address = any(e.recipients)) as received,
       count(e.id) filter (where e.type = 'message-ingest.spam' and a.address = any(e.recipients)) as junk,
       count(e.id) filter (where e.type = 'queue.authenticated-message-queued' and e.sender = a.address) as replies_sent,
       count(e.id) filter (where e.severity = 'error') as errors,
       latest.last_activity
     from unnest($2::text[]) as a(address)
     left join events e on e.addresses @> array[a.address] and e.event_at >= $1
     left join lateral (
       select max(l.event_at) as last_activity from events l where l.addresses @> array[a.address]
     ) latest on true
     group by a.address, latest.last_activity
     order by a.address`,
    [params.since, [...params.agentAddresses]],
  );
  return result.rows.map((row) => ({
    address: row.address as string,
    received: Number(row.received),
    junk: Number(row.junk),
    repliesSent: Number(row.replies_sent),
    errors: Number(row.errors),
    lastActivity: row.last_activity ? (row.last_activity as Date).toISOString() : null,
  }));
}
```

```ts
import { BatchError, normalizeEvent, parseBatch, type EventRow } from "./normalize";
import { verifySignature } from "./signature";

export const MAX_INGEST_BYTES = 5 * 1024 * 1024;
const PRUNE_INTERVAL_MS = 60 * 60 * 1000;

export interface IngestDeps {
  signingKey: string;
  insert: (rows: EventRow[]) => Promise<number>;
  prune: () => Promise<number>;
  now: () => number;
  log: (entry: Record<string, unknown>) => void;
}

/** Handles the server's event webhook: size limit, signature, parse, store, occasional prune. */
export function createIngestHandler(deps: IngestDeps): (request: Request) => Promise<Response> {
  let lastPruneAt = Number.NEGATIVE_INFINITY;
  return async (request) => {
    const body = await readBody(request, MAX_INGEST_BYTES);
    if (body === null) return reply(413);
    if (!verifySignature(body, request.headers.get("x-signature"), deps.signingKey)) {
      deps.log({ severity: "WARNING", message: "console.ingest.bad-signature" });
      return reply(401);
    }

    let rows: EventRow[];
    try {
      rows = parseBatch(new TextDecoder().decode(body)).map(normalizeEvent);
    } catch (error) {
      if (error instanceof BatchError) return reply(400);
      throw error;
    }

    try {
      await deps.insert(rows);
      const now = deps.now();
      if (now - lastPruneAt >= PRUNE_INTERVAL_MS) {
        lastPruneAt = now;
        await deps.prune();
      }
    } catch (error) {
      deps.log({
        severity: "ERROR",
        message: "console.ingest.store-failed",
        error: error instanceof Error ? error.message : String(error),
      });
      return reply(503);
    }
    return reply(204);
  };
}

function reply(status: number): Response {
  return new Response(null, { status, headers: { "Cache-Control": "no-store" } });
}

async function readBody(request: Request, limit: number): Promise<Uint8Array | null> {
  if (Number(request.headers.get("content-length") ?? "0") > limit) return null;
  if (!request.body) return new Uint8Array();
  const reader = request.body.getReader();
  const chunks: Uint8Array[] = [];
  let size = 0;
  for (;;) {
    const { done, value } = await reader.read();
    if (done) break;
    size += value.byteLength;
    if (size > limit) {
      await reader.cancel();
      return null;
    }
    chunks.push(value);
  }
  const body = new Uint8Array(size);
  let offset = 0;
  for (const chunk of chunks) {
    body.set(chunk, offset);
    offset += chunk.byteLength;
  }
  return body;
}
```

```ts
export const JMAP_CORE = "urn:ietf:params:jmap:core";
export const JMAP_MAIL = "urn:ietf:params:jmap:mail";
export const STALWART_REGISTRY = "urn:stalwart:jmap";

/** The only methods the console sends. Anything else is refused before a request is made. */
export const READ_ONLY_METHODS: ReadonlySet<string> = new Set([
  "Mailbox/get",
  "Email/query",
  "Email/get",
  "Thread/get",
  "x:Domain/query",
  "x:Domain/get",
  "x:Account/query",
  "x:Account/get",
]);

export type Invocation = [name: string, args: Record<string, unknown>, callId: string];

export interface Credentials {
  username: string;
  password: string;
}

export interface JmapSession {
  username: string;
  apiUrl: string;
  primaryAccounts: Record<string, string>;
}

export class JmapError extends Error {
  constructor(
    message: string,
    readonly type: string,
    readonly status?: number,
  ) {
    super(message);
    this.name = "JmapError";
  }
}

/** The server's master-user login: act as `mailbox` with the password of `principal`. */
export function masterUserLogin(mailbox: string, principal: string): string {
  return `${mailbox}%${principal}`;
}

export function responseOf(responses: readonly Invocation[], callId: string): Record<string, unknown> {
  const response = responses.find(([, , id]) => id === callId);
  if (!response) throw new JmapError(`the server sent no response for call ${callId}`, "missingResponse");
  return response[1];
}

type Fetch = (input: string | URL, init?: RequestInit) => Promise<Response>;

const SESSION_TTL_MS = 10 * 60 * 1000;
const REQUEST_TIMEOUT_MS = 15_000;

export class JmapClient {
  private readonly sessions = new Map<string, { session: JmapSession; expiresAt: number }>();

  constructor(
    private readonly baseUrl: string,
    private readonly fetchImpl: Fetch = fetch,
    private readonly now: () => number = Date.now,
  ) {}

  async session(credentials: Credentials): Promise<JmapSession> {
    const cached = this.sessions.get(credentials.username);
    if (cached && cached.expiresAt > this.now()) return cached.session;
    const session = (await this.request(credentials, new URL("/jmap/session", this.baseUrl), { method: "GET" })) as JmapSession;
    if (typeof session?.apiUrl !== "string" || typeof session.primaryAccounts !== "object") {
      throw new JmapError("the mail server returned an invalid session", "invalidSession");
    }
    this.sessions.set(credentials.username, { session, expiresAt: this.now() + SESSION_TTL_MS });
    return session;
  }

  async call(credentials: Credentials, using: readonly string[], calls: readonly Invocation[]): Promise<Invocation[]> {
    for (const [name] of calls) {
      if (!READ_ONLY_METHODS.has(name)) {
        throw new JmapError(`refusing to send ${name}: the console is read-only`, "forbiddenMethod");
      }
    }
    const session = await this.session(credentials);
    // The session advertises the server's public URL; keep its path but use the configured origin.
    const apiUrl = new URL(new URL(session.apiUrl).pathname, this.baseUrl);
    const body = (await this.request(credentials, apiUrl, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ using, methodCalls: calls }),
    })) as { methodResponses?: Invocation[] };
    if (!Array.isArray(body?.methodResponses)) {
      throw new JmapError("the mail server returned no method responses", "invalidResponse");
    }
    for (const [name, args] of body.methodResponses) {
      if (name === "error") {
        const description = typeof args.description === "string" && args.description ? `: ${args.description}` : "";
        throw new JmapError(`${String(args.type)}${description}`, String(args.type));
      }
    }
    return body.methodResponses;
  }

  private async request(credentials: Credentials, url: URL, init: RequestInit): Promise<unknown> {
    let response: Response;
    try {
      response = await this.fetchImpl(url, {
        ...init,
        headers: { ...(init.headers as Record<string, string> | undefined), Authorization: basicAuth(credentials) },
        cache: "no-store",
        signal: AbortSignal.timeout(REQUEST_TIMEOUT_MS),
      });
    } catch (error) {
      throw new JmapError(`cannot reach the mail server: ${error instanceof Error ? error.message : String(error)}`, "network");
    }
    if (!response.ok) {
      if (response.status === 401) this.sessions.delete(credentials.username);
      throw new JmapError(
        `the mail server answered HTTP ${response.status}`,
        response.status === 401 ? "unauthorized" : "http",
        response.status,
      );
    }
    return response.json();
  }
}

function basicAuth(credentials: Credentials): string {
  return `Basic ${Buffer.from(`${credentials.username}:${credentials.password}`).toString("base64")}`;
}
```

```ts
import { JMAP_CORE, STALWART_REGISTRY, responseOf, type Credentials, type JmapClient } from "./jmap";

export interface Agent {
  address: string;
  /** The account's description, which the agent runtime sets to the agent's name. */
  name: string | null;
}

export interface DirectorySettings {
  agentDomains: readonly string[];
  agentLocalpartPattern: RegExp | null;
  principal: string;
}

export function isAgentAddress(address: string, settings: DirectorySettings): boolean {
  const normalized = address.toLowerCase();
  const at = normalized.lastIndexOf("@");
  if (at <= 0 || normalized === settings.principal) return false;
  const local = normalized.slice(0, at);
  const domain = normalized.slice(at + 1);
  return settings.agentDomains.includes(domain) && (settings.agentLocalpartPattern?.test(local) ?? true);
}

/** Every user account whose address is an agent address. The registry ignores paging, so each domain is one query. */
export async function fetchAgents(
  client: Pick<JmapClient, "call">,
  credentials: Credentials,
  settings: DirectorySettings,
): Promise<Agent[]> {
  const using = [JMAP_CORE, STALWART_REGISTRY];
  const domainResponses = await client.call(credentials, using, [
    ["x:Domain/query", {}, "dq"],
    ["x:Domain/get", { "#ids": { resultOf: "dq", name: "x:Domain/query", path: "/ids" }, properties: ["name"] }, "dg"],
  ]);
  const domains = (responseOf(domainResponses, "dg").list as { id: string; name: string }[]).filter((domain) =>
    settings.agentDomains.includes(domain.name.toLowerCase()),
  );

  const agents: Agent[] = [];
  for (const domain of domains) {
    const responses = await client.call(credentials, using, [
      ["x:Account/query", { filter: { domainId: domain.id } }, "aq"],
      [
        "x:Account/get",
        { "#ids": { resultOf: "aq", name: "x:Account/query", path: "/ids" }, properties: ["name", "description", "@type"] },
        "ag",
      ],
    ]);
    const accounts = responseOf(responses, "ag").list as { name: string; description?: string | null; "@type"?: string }[];
    for (const account of accounts) {
      if (account["@type"] !== undefined && account["@type"] !== "User") continue;
      const address = `${account.name}@${domain.name}`.toLowerCase();
      if (isAgentAddress(address, settings)) {
        agents.push({ address, name: account.description?.trim() || null });
      }
    }
  }
  return agents.sort((a, b) => (a.name ?? a.address).localeCompare(b.name ?? b.address) || a.address.localeCompare(b.address));
}
```

```ts
const LAYOUT_ATTRIBUTES = [
  "style", "align", "valign", "width", "height", "bgcolor", "color", "dir", "lang",
  "colspan", "rowspan", "border", "cellpadding", "cellspacing",

/** Safe markup for display: no scripts, handlers, forms or frames, and no remote images. */
export function sanitizeEmailHtml(html: string): string {
  return sanitizeHtml(html, EMAIL_HTML);
}

const BLOCK_END = /<\/(?:p|div|tr|li|h[1-6]|blockquote|table)>|<br\s*\/?>/gi;
const NAMED_ENTITIES: Record<string, string> = { amp: "&", lt: "<", gt: ">", quot: '"', apos: "'", nbsp: " " };

export function decodeEntities(text: string): string {
  return text.replace(/&(#x[0-9a-f]+|#\d+|[a-z]+);/gi, (match, entity: string) => {
    if (entity.startsWith("#")) {
      const code = entity[1]?.toLowerCase() === "x" ? Number.parseInt(entity.slice(2), 16) : Number.parseInt(entity.slice(1), 10);
      return Number.isInteger(code) && code > 0 && code <= 0x10ffff ? String.fromCodePoint(code) : match;
    }
    return NAMED_ENTITIES[entity.toLowerCase()] ?? match;
  });
}

/** Plain text for an HTML-only message. */
export function htmlToText(html: string): string {
  const stripped = sanitizeHtml(html.replace(BLOCK_END, (tag) => `${tag}\n`), { allowedTags: [], allowedAttributes: {} });
  return decodeEntities(stripped)
    .replace(/ /g, " ")
    .replace(/[ \t]+\n/g, "\n")
    .replace(/\n{3,}/g, "\n\n")
    .trim();
}

/** A document for `<iframe sandbox srcdoc>`: its own policy blocks every remote resource. */
export function emailHtmlDocument(sanitizedHtml: string): string {
  return [
    "<!doctype html><html><head><meta charset=\"utf-8\">",
    "<meta http-equiv=\"Content-Security-Policy\" content=\"default-src 'none'; style-src 'unsafe-inline'; img-src data:; font-src data:\">",
    "<style>body{font-family:system-ui,sans-serif;font-size:14px;line-height:1.5;color:#292929;margin:12px;overflow-wrap:anywhere}</style>",
    `</head><body>${sanitizedHtml}</body></html>`,
  ].join("");
}
```

```ts
export function rolesFromMailboxes(mailboxes: readonly { id: string; role: string | null }[]): MailboxRoles {

export function directionOf(email: JmapEmail, agentAddress: string, roles: MailboxRoles): Direction {
  if (roles.sent !== null && email.mailboxIds[roles.sent]) return "outbound";
  return email.from?.[0]?.email.toLowerCase() === agentAddress ? "outbound" : "inbound";

  return Boolean((roles.junk !== null && email.mailboxIds[roles.junk]) || email.keywords?.$junk);

/** Machine-generated mail (RFC 3834), including delivery reports, never awaits a reply. */
export function isAutomated(email: JmapEmail): boolean {
  const autoSubmitted = email["header:Auto-Submitted:asText"]?.trim().toLowerCase();
  if (autoSubmitted && autoSubmitted !== "no") return true;
  const local = email.from?.[0]?.email.split("@")[0]?.toLowerCase();
  return local === "mailer-daemon" || local === "postmaster";
}

/** `latest` holds the newest email of each thread, in display order. */
export function summarizeThreads(
  latest: readonly JmapEmail[],
  threads: readonly { id: string; emailIds: string[] }[],
  agentAddress: string,
  roles: MailboxRoles,
): ThreadSummary[] {
  const counts = new Map(threads.map((thread) => [thread.id, thread.emailIds.length]));
  return latest.map((email) => {
    const direction = directionOf(email, agentAddress, roles);
    const counterpart = direction === "inbound" ? email.from?.[0] : email.to?.[0];
    let status: ThreadStatus = "answered";
    if (direction === "inbound") {
      if (isJunk(email, roles)) status = "junk";
      else if (isAutomated(email)) status = "automated";
      else status = "awaiting-reply";
    }
    return {
      threadId: email.threadId,
      subject: email.subject?.trim() || "(no subject)",
      counterpart: counterpart ? lowercase(counterpart) : null,
      messageCount: counts.get(email.threadId) ?? 1,
      lastActivity: email.receivedAt,
      lastDirection: direction,
      preview: email.preview ?? "",
      status,
    };
  });
}

function partValues(email: JmapEmail, parts: readonly BodyPart[]): string[] {
  return parts
    .map((part) => (part.partId === null ? undefined : email.bodyValues?.[part.partId]?.value))
    .filter((value): value is string => typeof value === "string");
}

export function buildThread(
  emails: readonly JmapEmail[],
  agentAddress: string,
  roles: MailboxRoles,
  renderers: BodyRenderers,
): ThreadMessage[] {
  return [...emails]
    .sort((a, b) => a.receivedAt.localeCompare(b.receivedAt))
    .map((email) => {
      const htmlParts = (email.htmlBody ?? []).filter((part) => part.type === "text/html");
      const html = htmlParts.length > 0 ? renderers.sanitizeHtml(partValues(email, htmlParts).join("\n")) : null;
      let text = partValues(email, email.textBody ?? []).join("\n\n").trim();
      if (!text && html) text = renderers.htmlToText(html);
      const direction = directionOf(email, agentAddress, roles);
      return {
        id: email.id,
        direction,
        subject: email.subject?.trim() || "(no subject)",
        from: (email.from ?? []).map(lowercase),
        to: (email.to ?? []).map(lowercase),
        cc: (email.cc ?? []).map(lowercase),
        receivedAt: email.receivedAt,
        text,
        html,
        truncated: Object.values(email.bodyValues ?? {}).some((value) => value.isTruncated === true),
        junk: direction === "inbound" && isJunk(email, roles),
        automated: direction === "inbound" && isAutomated(email),
        attachments: (email.attachments ?? []).map((part) => ({ name: part.name ?? "(unnamed)", size: part.size, type: part.type })),
      };
    });
}
```

```ts
import { createRemoteJWKSet, jwtVerify, type JWTPayload, type JWTVerifyGetKey } from "jose";
import type { ConsoleConfig } from "./config";

export const IAP_ISSUER = "https://cloud.google.com/iap";
export const IAP_ASSERTION_HEADER = "x-goog-iap-jwt-assertion";

export interface Operator {
  email: string;
}

export class AuthError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "AuthError";
  }
}

type AuthSettings = Pick<ConsoleConfig, "authMode" | "iapAudience" | "iapJwksUrl" | "allowedEmails" | "allowedDomains">;

let remoteKeys: { url: string; keys: JWTVerifyGetKey } | undefined;

function keysFor(url: string): JWTVerifyGetKey {
  if (remoteKeys?.url !== url) remoteKeys = { url, keys: createRemoteJWKSet(new URL(url)) };
  return remoteKeys.keys;
}

export function isAllowed(email: string, allowedEmails: readonly string[], allowedDomains: readonly string[]): boolean {
  const normalized = email.toLowerCase();
  const domain = normalized.slice(normalized.lastIndexOf("@") + 1);
  return allowedEmails.includes(normalized) || allowedDomains.includes(domain);
}

/** Resolves the operator from the identity-aware proxy's signed assertion; throws AuthError otherwise. */
export async function authenticate(headers: Headers, settings: AuthSettings, keys?: JWTVerifyGetKey): Promise<Operator> {
  if (settings.authMode === "none") return { email: "local-operator" };
  const token = headers.get(IAP_ASSERTION_HEADER);
  if (!token) throw new AuthError("no identity assertion");

  let payload: JWTPayload;
  try {
    ({ payload } = await jwtVerify(token, keys ?? keysFor(settings.iapJwksUrl), {
      issuer: IAP_ISSUER,
      audience: settings.iapAudience,
      algorithms: ["ES256"],
    }));
  } catch (error) {
    throw new AuthError(`invalid identity assertion: ${error instanceof Error ? error.message : String(error)}`);
  }

  const email = typeof payload.email === "string" ? payload.email.toLowerCase() : "";
  if (!email) throw new AuthError("the identity assertion has no email");
  if (!isAllowed(email, settings.allowedEmails, settings.allowedDomains)) throw new AuthError(`${email} is not allowed`);
  return { email };
}
```

```ts
import { NextResponse, type NextRequest } from "next/server";
import { AuthError, authenticate } from "@/lib/auth";
import { getConfig } from "@/lib/config";
import { gateFor } from "@/lib/gate";

export async function proxy(request: NextRequest): Promise<NextResponse> {
  const settings = getConfig();
  const decision = gateFor(request.nextUrl.pathname, settings);
  if (decision === "pass") return NextResponse.next();
  if (decision === "not-found") return new NextResponse(null, { status: 404 });
  try {
    await authenticate(request.headers, settings);
    return NextResponse.next();
  } catch (error) {
    if (error instanceof AuthError) {
      return new NextResponse("Forbidden", { status: 403, headers: { "Content-Type": "text/plain; charset=utf-8" } });
    }
    throw error;
  }
}
```

```ts
async function openMailbox(client: MailboxClient, principal: Credentials, agentAddress: string): Promise<OpenMailbox> {
  const credentials = { username: masterUserLogin(agentAddress, principal.username), password: principal.password };
  const session = await client.session(credentials);
  const accountId = session.primaryAccounts[JMAP_MAIL];
  if (!accountId) throw new JmapError("the mailbox has no mail account", "accountNotFound");
  const responses = await client.call(credentials, [JMAP_CORE, JMAP_MAIL], [
    ["Mailbox/get", { accountId, ids: null, properties: ["name", "role"] }, "m"],
  ]);
  const roles = rolesFromMailboxes(responseOf(responses, "m").list as { id: string; role: string | null }[]);
  return { credentials, accountId, roles };

/** One page of the agent's threads, newest first, without Trash and Drafts. */
export async function listThreads(client: MailboxClient, principal: Credentials, agentAddress: string, position: number): Promise<ThreadPage> {
  const mailbox = await openMailbox(client, principal, agentAddress);
  const hidden = [mailbox.roles.trash, mailbox.roles.drafts].filter((id): id is string => id !== null);
  const { accountId } = mailbox;
  const responses = await client.call(mailbox.credentials, [JMAP_CORE, JMAP_MAIL], [
    [
      "Email/query",
      {
        accountId,
        filter: hidden.length > 0 ? { inMailboxOtherThan: hidden } : {},
        sort: [{ property: "receivedAt", isAscending: false }],
        collapseThreads: true,
        position,
        limit: THREAD_PAGE_SIZE,
        calculateTotal: true,
      },
      "q",
    ],
    ["Email/get", { accountId, "#ids": { resultOf: "q", name: "Email/query", path: "/ids" }, properties: SUMMARY_PROPERTIES }, "g"],
    ["Thread/get", { accountId, "#ids": { resultOf: "g", name: "Email/get", path: "/list/*/threadId" } }, "t"],
  ]);
  const query = responseOf(responses, "q");
  const order = new Map((query.ids as string[]).map((id, index) => [id, index]));
  const latest = [...(responseOf(responses, "g").list as JmapEmail[])].sort(
    (a, b) => (order.get(a.id) ?? 0) - (order.get(b.id) ?? 0),
  );
  const threads = responseOf(responses, "t").list as { id: string; emailIds: string[] }[];
  return {
    threads: summarizeThreads(latest, threads, agentAddress, mailbox.roles),
    total: typeof query.total === "number" ? query.total : latest.length,
    position,
  };
}

/** Every message of one thread, oldest first; null when the thread is not in this mailbox. */
export async function getThread(
  client: MailboxClient,
  principal: Credentials,
  agentAddress: string,
  threadId: string,
): Promise<ThreadMessage[] | null> {
  const mailbox = await openMailbox(client, principal, agentAddress);
  const { accountId } = mailbox;
  const responses = await client.call(mailbox.credentials, [JMAP_CORE, JMAP_MAIL], [
    ["Thread/get", { accountId, ids: [threadId] }, "t"],
    [
      "Email/get",
      {
        accountId,
        "#ids": { resultOf: "t", name: "Thread/get", path: "/list/*/emailIds" },
        properties: MESSAGE_PROPERTIES,
        fetchTextBodyValues: true,
        fetchHTMLBodyValues: true,
        maxBodyValueBytes: MAX_BODY_BYTES,
      },
      "e",
    ],
  ]);
  if ((responseOf(responses, "t").list as unknown[]).length === 0) return null;
  const emails = responseOf(responses, "e").list as JmapEmail[];
  return buildThread(emails, agentAddress, mailbox.roles, { sanitizeHtml: sanitizeEmailHtml, htmlToText });
}
```

```ts
/** The agent directory, cached for five minutes. */
export async function getAgents(): Promise<Agent[]> {
  const now = Date.now();
  if (agentCache && now - agentCache.loadedAt < AGENT_CACHE_MS) return agentCache.agents;
  const agents = await fetchAgents(jmapClient(), principalCredentials(), directorySettings());
  agentCache = { loadedAt: now, agents };
  return agents;
}

  const normalized = address.toLowerCase();
  return (await getAgents()).find((agent) => agent.address === normalized);
```

```ts
  let formatter = formatters.get(timeZone);
  if (!formatter) {
    formatter = new Intl.DateTimeFormat("en-GB", {
      timeZone,
      day: "2-digit",
      month: "short",
      year: "numeric",
      hour: "2-digit",
      minute: "2-digit",
      second: "2-digit",
      hourCycle: "h23",
    });
    formatters.set(timeZone, formatter);
  }
  return formatter.format(typeof value === "string" ? new Date(value) : value);
}

export function formatRelative(value: string | Date, now: Date): string {
  const seconds = Math.round((now.getTime() - new Date(value).getTime()) / 1000);
  if (seconds < 45) return "just now";
  const minutes = Math.round(seconds / 60);
  if (minutes < 60) return `${minutes} min ago`;
  const hours = Math.round(minutes / 60);
  if (hours < 24) return `${hours} h ago`;
  return `${Math.round(hours / 24)} d ago`;
}

export function formatBytes(bytes: number): string {
  if (bytes < 1024) return `${bytes} B`;
  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
  return `${(bytes / 1024 / 1024).toFixed(1)} MB`;
}

export function formatAddress(address: EmailAddress | null | undefined): string {
  if (!address) return "unknown";
  return address.name ? `${address.name} <${address.email}>` : address.email;
}

/** One line for an event: who to whom, then whatever the server said about it. */
export function describeEvent(event: { sender: string | null; recipients: string[]; data: Record<string, unknown> }): string {
  const parts: string[] = [];
  if (event.sender || event.recipients.length > 0) {
    parts.push(`${event.sender ?? "—"} → ${event.recipients.join(", ") || "—"}`);
  }
  const detail = event.data.details ?? event.data.reason;
  if (typeof detail === "string" && detail) parts.push(detail);
  if (typeof event.data.code === "number") parts.push(`code ${event.data.code}`);
  if (typeof event.data.remoteIp === "string" && !event.sender && event.recipients.length === 0) {
    parts.push(`from ${event.data.remoteIp}`);
  }
  return parts.join(" · ");
}
```

```ts
import type { Severity } from "./eventTypes";

export const EVENT_WINDOWS = { "1h": 3_600_000, "24h": 86_400_000, "7d": 604_800_000, "30d": 2_592_000_000 } as const;
export type EventWindow = keyof typeof EVENT_WINDOWS;
export const EVENT_TYPE_PREFIXES = ["message-ingest.", "queue.", "delivery.", "smtp.", "auth.", "security."] as const;

export interface EventFilters {
  typePrefix?: string;
  severity?: Severity;
  address?: string;
  window: EventWindow;
  since: Date;
}

interface ParamReader {
  get(name: string): string | null;
}

export function parseEventFilters(params: ParamReader, now: Date): EventFilters {
  const filters: EventFilters = { window: "24h", since: new Date(0) };
  const type = params.get("type")?.trim() ?? "";
  if (/^[a-z0-9.-]{1,80}$/.test(type)) filters.typePrefix = type;
  const severity = params.get("severity");
  if (severity === "info" || severity === "warning" || severity === "error") filters.severity = severity;
  const agent = params.get("agent")?.trim().toLowerCase() ?? "";
  if (/^[^\s@]+@[^\s@]+$/.test(agent)) filters.address = agent;
  const window = params.get("window") ?? "";
  if (window in EVENT_WINDOWS) filters.window = window as EventWindow;
  filters.since = new Date(now.getTime() - EVENT_WINDOWS[filters.window]);
  return filters;
}

/** The filters as a query string, leaving out defaults. */
export function eventFiltersQuery(filters: Omit<EventFilters, "since">): string {
  const params = new URLSearchParams();
  if (filters.typePrefix) params.set("type", filters.typePrefix);
  if (filters.severity) params.set("severity", filters.severity);
  if (filters.address) params.set("agent", filters.address);
  if (filters.window !== "24h") params.set("window", filters.window);
  return params.toString();
}
```

```ts
import type { NextRequest } from "next/server";
import { AuthError, authenticate } from "@/lib/auth";
import { getConfig, servesWeb } from "@/lib/config";
import { getPool } from "@/lib/db";
import { parseEventFilters } from "@/lib/eventFilters";
import { queryEvents } from "@/lib/eventStore";

export async function GET(request: NextRequest): Promise<Response> {
  const config = getConfig();
  if (!servesWeb(config)) return new Response(null, { status: 404 });
  try {
    await authenticate(request.headers, config);
  } catch (error) {
    if (error instanceof AuthError) return Response.json({ error: "forbidden" }, { status: 403 });
    throw error;
  }
  const params = request.nextUrl.searchParams;
  const after = Number(params.get("after"));
  if (!Number.isSafeInteger(after) || after < 0) {
    return Response.json({ error: "after must be a non-negative integer" }, { status: 400 });
  }
  const filters = parseEventFilters(params, new Date());
  const events = await queryEvents(getPool(), { ...filters, afterId: after, limit: 200 });
  return Response.json({ events }, { headers: { "Cache-Control": "no-store" } });
}
```

```tsx
"use client";

import { useEffect, useRef, useState } from "react";
import type { StoredEvent } from "@/lib/eventStore";
import { EventRow } from "./EventRow";

const POLL_MS = 3000;
const KEEP = 500;

export function LiveEvents({ initialAfter, query, timeZone }: { initialAfter: number; query: string; timeZone: string }) {
  const [live, setLive] = useState(false);
  const [events, setEvents] = useState<StoredEvent[]>([]);
  const [error, setError] = useState<string | null>(null);
  const after = useRef(initialAfter);

  useEffect(() => {
    if (!live) return;
    let cancelled = false;
    const poll = async () => {
      try {
        const response = await fetch(`/api/events?${query ? `${query}&` : ""}after=${after.current}`, { cache: "no-store" });
        if (!response.ok) throw new Error(`HTTP ${response.status}`);
        const body = (await response.json()) as { events: StoredEvent[] };
        if (cancelled) return;
        setError(null);
        if (body.events.length > 0) {
          after.current = Math.max(after.current, ...body.events.map((event) => event.id));
          setEvents((current) => [...body.events, ...current].slice(0, KEEP));
        }
      } catch (caught) {
        if (!cancelled) setError(caught instanceof Error ? caught.message : String(caught));
      }
    };
    void poll();
    const timer = setInterval(poll, POLL_MS);
    return () => {
      cancelled = true;
      clearInterval(timer);
    };
  }, [live, query]);

  return (
    <section aria-label="Live events" className="overflow-hidden rounded-lg border border-border bg-surface">
      <div className="flex items-center gap-3 px-4 py-3 text-sm">
        <button
          type="button"
          onClick={() => setLive((value) => !value)}
          aria-pressed={live}
          className={`rounded-md px-3 py-1.5 text-xs font-medium ${live ? "bg-brand text-surface hover:bg-brand-strong" : "border border-border hover:bg-surface-muted"}`}
        >
          {live ? "Stop live updates" : "Start live updates"}
        </button>
        <span className="text-xs text-text-muted">
          {live ? (error ? `Paused: ${error}. Retrying…` : `Checking every ${POLL_MS / 1000} seconds.`) : "New events appear here while live updates are on."}
        </span>
        {events.length > 0 ? <span className="ml-auto text-xs font-medium">{events.length} new</span> : null}
      </div>
      {events.length > 0 ? (
        <div className="overflow-x-auto">
          <table className="w-full">
            <tbody>
              {events.map((event) => (
                <EventRow key={event.id} event={event} timeZone={timeZone} />
              ))}
            </tbody>
          </table>
        </div>
      ) : null}
    </section>
  );
```

```js
// Creates or updates the console principal and the event webhook on an inWorkspace server.
// Idempotent; prints no secret. See console/README.md#server-setup.
import { readFileSync } from "node:fs";
import { pathToFileURL } from "node:url";

const eventTypes = JSON.parse(readFileSync(new URL("../src/lib/event-types.json", import.meta.url), "utf8"));

const PRINCIPAL_PERMISSIONS = {
  "@type": "Merge",
  enabledPermissions: { impersonate: true, sysAccountGet: true, sysAccountQuery: true, sysDomainGet: true, sysDomainQuery: true },
  disabledPermissions: {},
};

export class RegistryClient {
  constructor(baseUrl, username, password) {
    this.baseUrl = baseUrl.replace(/\/+$/, "");
    this.authorization = `Basic ${Buffer.from(`${username}:${password}`).toString("base64")}`;
  }

  async call(methodCalls) {
    const response = await fetch(`${this.baseUrl}/jmap/`, {
      method: "POST",
      headers: { Authorization: this.authorization, "Content-Type": "application/json" },
      body: JSON.stringify({ using: ["urn:ietf:params:jmap:core", "urn:stalwart:jmap"], methodCalls }),
    });
    if (!response.ok) throw new Error(`registry request failed with HTTP ${response.status}`);
    const body = await response.json();
    for (const [name, args] of body.methodResponses) {
      if (name === "error") throw new Error(`registry error ${args.type}${args.description ? `: ${args.description}` : ""}`);
    }
    return body.methodResponses;
  }

  async one(name, args) {
    const [[, result]] = await this.call([[name, args, "0"]]);
    return result;
  }
}

function createdId(result, key, what) {
  if (result.notCreated?.[key]) throw new Error(`could not create ${what}: ${JSON.stringify(result.notCreated[key])}`);
  return result.created[key].id;
}

function assertUpdated(result, id, what) {
  if (result.notUpdated?.[id]) throw new Error(`could not update ${what}: ${JSON.stringify(result.notUpdated[id])}`);
}

export async function domainId(client, name) {
  const result = await client.one("x:Domain/query", { filter: { name: name.toLowerCase() } });
  return result.ids[0] ?? null;
}

export async function ensureAccount(client, { address, description, secret, permissions }) {
  const [local, domain] = address.toLowerCase().split("@");
  const id = await domainId(client, domain);
  if (!id) throw new Error(`the domain ${domain} does not exist on the server`);
  const responses = await client.call([
    ["x:Account/query", { filter: { domainId: id } }, "q"],
    ["x:Account/get", { "#ids": { resultOf: "q", name: "x:Account/query", path: "/ids" }, properties: ["name"] }, "g"],
  ]);
  const existing = responses[1][1].list.find((account) => account.name.toLowerCase() === local);
  const fields = {
    description,
    ...(secret ? { credentials: { 0: { "@type": "Password", secret } } } : {}),
    ...(permissions ? { permissions } : {}),
  };
  if (existing) {
    assertUpdated(await client.one("x:Account/set", { update: { [existing.id]: fields } }), existing.id, address);
    return { id: existing.id, created: false };
  }
  const result = await client.one("x:Account/set", { create: { a: { "@type": "User", name: local, domainId: id, ...fields } } });
  return { id: createdId(result, "a", address), created: true };
}

export function ensurePrincipal(client, { address, secret }) {
  return ensureAccount(client, {
    address,
    secret,
    description: "inWorkspace Console (read-only oversight)",
    permissions: PRINCIPAL_PERMISSIONS,
  });
}

export async function ensureWebhook(client, { url, signingKey }) {
  const responses = await client.call([
    ["x:WebHook/query", {}, "q"],
    ["x:WebHook/get", { "#ids": { resultOf: "q", name: "x:WebHook/query", path: "/ids" }, properties: ["url"] }, "g"],
  ]);
  const existing = responses[1][1].list.find((hook) => hook.url === url);
  const fields = {
    url,
    signatureKey: { "@type": "Value", secret: signingKey },
    enable: true,
    level: "info",
    lossy: false,
    eventsPolicy: "include",
    events: Object.fromEntries(eventTypes.subscribed.map((type) => [type, true])),
    throttle: 1000,
    timeout: 30000,
    discardAfter: 3600000,
  };
  if (existing) {
    assertUpdated(await client.one("x:WebHook/set", { update: { [existing.id]: fields } }), existing.id, "the webhook");
    return { id: existing.id, created: false };
  }
  const result = await client.one("x:WebHook/set", { create: { w: fields } });
  return { id: createdId(result, "w", "the webhook"), created: true };
}

/** Some server versions refuse to reload because of an unrelated default setting; the webhook then activates on restart. */
export async function reloadSettings(client) {
  try {
    const result = await client.one("x:Action/set", { create: { r: { "@type": "ReloadSettings" } } });
    if (result.notCreated?.r) return { reloaded: false, reason: JSON.stringify(result.notCreated.r) };
    return { reloaded: true };
  } catch (error) {
    return { reloaded: false, reason: error.message };
  }
}

function required(name) {
  const value = process.env[name]?.trim();
  if (!value) throw new Error(`${name} is required`);
  return value;
}

async function main() {
  const client = new RegistryClient(required("SERVER_URL"), required("SERVER_ADMIN_USER"), required("SERVER_ADMIN_SECRET"));
  const principal = await ensurePrincipal(client, {
    address: required("CONSOLE_PRINCIPAL"),
    secret: required("CONSOLE_PRINCIPAL_SECRET"),
  });
  console.log(`console principal ${principal.created ? "created" : "updated"} (id ${principal.id})`);
  const webhook = await ensureWebhook(client, { url: required("WEBHOOK_URL"), signingKey: required("WEBHOOK_SIGNING_KEY") });
  console.log(`webhook ${webhook.created ? "created" : "updated"} (id ${webhook.id}, ${eventTypes.subscribed.length} event types)`);
  const reload = await reloadSettings(client);
  console.log(reload.reloaded ? "settings reloaded" : `settings not reloaded (${reload.reason}); restart the server to activate the webhook`);
}

if (process.argv[1] && import.meta.url === pathToFileURL(process.argv[1]).href) {
  main().catch((error) => {

    process.exit(1);
  });
}
```

```js
// A minimal SMTP client for the end-to-end run: one message, plain text, no TLS.
import net from "node:net";

export function sendMail({ host, port, from, to, data }) {
  const commands = [
    "EHLO e2e.example.com",
    `MAIL FROM:<${from}>`,
    `RCPT TO:<${to}>`,
    "DATA",
    `${data.replace(/\r?\n/g, "\r\n").replace(/^\./gm, "..")}\r\n.`,
    "QUIT",
  ];
  return new Promise((resolve, reject) => {
    const socket = net.createConnection({ host, port });
    let buffer = "";
    let step = -1;
    let refused = null;
    socket.setTimeout(30000, () => socket.destroy(new Error("SMTP timed out")));
    socket.on("error", reject);
    socket.on("close", () => resolve(refused ? { accepted: false, reply: refused } : { accepted: true }));
    socket.on("data", (chunk) => {
      buffer += chunk.toString("utf8");
      let end;
      while ((end = buffer.indexOf("\r\n")) >= 0) {
        const line = buffer.slice(0, end);
        buffer = buffer.slice(end + 2);
        if (!/^\d{3} /.test(line)) continue;
        const code = Number(line.slice(0, 3));
        if (code >= 400 && !refused) {
          refused = line;
          step = commands.length - 2;
        }
        step += 1;
        if (step < commands.length) socket.write(`${commands[step]}\r\n`);
      }
    });
  });
}
```

```js
    const seen = rows.map((row) => row.type);
    const ingested = seen.some((type) => type.startsWith("message-ingest."));
    return ingested && expected.every((type) => seen.includes(type)) ? seen : null;
  });
  await db.end();
  console.log(`  ✓ ${types.length} event types stored: ${types.sort().join(", ")}`);

  step("Checking the console");
  expectText("the overview", await (await fetch(`${CONSOLE}/`)).text(), ["Front desk agent", "Concierge agent", "Replies sent"]);
  const agentPath = `/agents/${encodeURIComponent(AGENT)}`;
  expectText("the agent's conversations", await (await fetch(`${CONSOLE}${agentPath}`)).text(), ["Late check-in on Friday?", "Answered"]);
  expectText(
    "the thread",
    await (await fetch(`${CONSOLE}${agentPath}/threads/${inboundEmail.threadId}`)).text(),
    ["Hi, can I check in at 11pm on Friday?", "The night desk will have your key.", "Agent reply"],
  );
  expectText("the events page", await (await fetch(`${CONSOLE}/events?type=delivery.`)).text(), ["delivery.dsn-perm-fail", "null MX"]);
  const api = await (await fetch(`${CONSOLE}/api/events?after=0`)).json();
  if (!Array.isArray(api.events) || api.events.length === 0) throw new Error("the events API returned no events");
  console.log(`  ✓ the events API returns ${api.events.length} events`);
  const unsigned = await fetch(`${CONSOLE}/api/ingest/events`, { method: "POST", body: '{"events":[]}' });
  if (unsigned.status !== 401) throw new Error(`an unsigned webhook post got ${unsigned.status}, expected 401`);
  console.log("  ✓ an unsigned webhook post is refused with 401");
  // notFound() inside a streamed Suspense boundary keeps HTTP 200, so check what the page shows.
  const outside = await (await fetch(`${CONSOLE}/agents/${encodeURIComponent(`runtime@${DOMAIN}`)}`)).text();
  if (!outside.includes("There is no agent, thread or page at this address.") || outside.includes("Agent runtime")) {
    throw new Error("a mailbox outside the agent directory could be opened");
  }
  console.log("  ✓ a mailbox outside the agent directory is not opened");

  console.log("\nEnd-to-end run passed.");
  if (keep) {
    console.log(`Console: ${CONSOLE}  (Ctrl-C to stop; the containers keep running until \`docker compose -f e2e/compose.yaml down -v\`)`);
    await new Promise(() => {});
  }
} finally {
  if (!keep) {
    consoleProcess?.kill();
    docker(["down", "-v", "--remove-orphans"]);
  }
}
```
